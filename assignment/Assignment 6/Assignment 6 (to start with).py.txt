import csv

alpha = 0.5
gamma = 0.9

rt = 0
SM = 0
action = 0

state_values = []
accumulated_reward = 0

max_q = list()




# generate a random multi dimensions array, value will between 0 and 1
Q = np.random.rand(2,2,2,2,2,2,2)

#Initialize Q table by all zeros.it is 7 dimensions array
#Q = [[[[[[[random.random() for t in range(2)]for i in range(3)]for j in range(2)]for k in range(2)]for a in range(2)]for n in range(2)]for m in range(3)]
print('init Q -------> ',Q)
at = 0
st0 = 0
st1 = 0
st2 = 0
st3 = 0
st4 = 0
sm = 0


class RL_Robot(Robot):

    def update(self):
        global action,numIterations
        global rt
        global at,st0,st1,st2,st3,st4,sm
        ir_values = self.distance()  # distance range [0, 100]
        target = self.smell()         # angle range [-180, 180]
        
        self.just_eat = False
        #print('ir_values',ir_values)
        #print('target',target)

        if target > 180:
            target = target-360

        
        FIR = int(self.distance()[0] < 40)
        LIR = int(self.distance()[7] < 40)
        RIR = int(self.distance()[1] < 40)
        LLIR = int(self.distance()[6] < 40)
        RRIR = int(self.distance()[2] < 40)
        
        if(target > 30):
            SM = 2
        elif(target < -30):
            SM = 0
	else :
	    SM = 1
 
        #if random.randint(0,9) > 0:
        if(random.random() < 0.9):
            # default
            action = 0
            # select the action have the largest q value

            if(Q[0][FIR][LIR][RIR][LLIR][RRIR][SM] > Q[1][FIR][LIR][RIR][LLIR][RRIR][SM]):
                if(Q[0][FIR][LIR][RIR][LLIR][RRIR][SM] > Q[1][FIR][LIR][RIR][LLIR][RRIR][SM]):
		    action = 0
		else :
		    action = 2
	    else :

                if(Q[1][FIR][LIR][RIR][LLIR][RRIR][SM] > Q[2][FIR][LIR][RIR][LLIR][RRIR][SM]):
                    action = 1
		else :
		    action = 2
        # randomly
        else: 
            action = np.random.randint(0,3)
           

        # Move Forward
        if(action == 0):
            self.move(5)
            if self.stuck :
                rt = -2
            else :
                rt = 1

            Q[at][st0][st1][st2][st3][st4][sm] += alpha * (rt + gamma*self.max_q_table(FIR,LIR,RIR,LLIR,RRIR,SM)-Q[at][st0][st1][st2][st3][st4][sm])
            at = action
            st0 = FIR
            st1 = LIR
            st2 = RIR
            st3 = LLIR
            st4 = RRIR
            sm = SM

	# Turn right
        elif (action == 2):
            self.turn(20)
            rt = -0.1 - (abs(target)/180)

            Q[at][st0][st1][st2][st3][st4][sm] += alpha * (rt + gamma*self.max_q_table(FIR,LIR,RIR,LLIR,RRIR,SM)-Q[at][st0][st1][st2][st3][st4][sm])
            at = action
            st0 = FIR
            st1 = LIR
            st2 = RIR
            st3 = LLIR
            st4 = RRIR
            sm = SM

        # Turn left
        elif(action == 1):
            self.turn(-20)
            rt = -0.1 - (abs(target)/180)

            Q[at][st0][st1][st2][st3][st4][sm] += alpha * (rt + gamma*self.max_q_table(FIR,LIR,RIR,LLIR,RRIR,SM)-Q[at][st0][st1][st2][st3][st4][sm])
            at = action
            st0 = FIR
            st1 = LIR
            st2 = RIR
            st3 = LLIR
            st4 = RRIR
            sm = SM


if __name__ == '__main__':
    app = PySimbotApp(robot_cls=RL_Robot, num_robots=1,max_tick=5000,simulation_forever=True)
